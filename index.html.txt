    <script>
        // Variáveis Globais
        let carrinho = [];
        
        // Adicionar item ao carrinho
        function addItem(nome, preco) {
            carrinho.push({ nome, preco });
            atualizarCarrinho();
        }

        // Atualizar interface do carrinho
        function atualizarCarrinho() {
            const lista = document.getElementById('lista-carrinho');
            lista.innerHTML = '';
            let subtotal = 0;

            carrinho.forEach((item, index) => {
                subtotal += item.preco;
                lista.innerHTML += `
                    <div class="carrinho-item">
                        <span>${item.nome} - R$ ${item.preco.toFixed(2)}</span>
                        <button onclick="removerItem(${index})">X</button>
                    </div>
                `;
            });

            document.querySelector('.btn-carrinho').innerText = `🛒 Ver Carrinho (${carrinho.length})`;
            document.getElementById('subtotal').innerText = `R$ ${subtotal.toFixed(2)}`;
            calcularTotal();
        }

        // Remover item
        function removerItem(index) {
            carrinho.splice(index, 1);
            atualizarCarrinho();
        }

        // Calcular total com taxa
        function calcularTotal() {
            let subtotal = carrinho.reduce((acc, item) => acc + item.preco, 0);
            let taxa = parseFloat(document.getElementById('distancia').value);
            let total = subtotal + taxa;
            document.getElementById('taxa').innerText = `R$ ${taxa.toFixed(2)}`;
            document.getElementById('total').innerText = `R$ ${total.toFixed(2)}`;
        }

        // Abrir/Fechar Carrinho
        function abrirCarrinho() {
            document.getElementById('modalCarrinho').style.display = 'block';
        }
        document.getElementById('modalCarrinho').addEventListener('click', function(e) {
            if (e.target === this) this.style.display = 'none';
        });

        // Buscar CEP na API ViaCEP (Simulação automática de endereço)
        async function buscarCep() {
            let cep = document.getElementById('cep').value.replace(/\D/g, '');
            if (cep.length !== 8) return alert('CEP inválido!');
            
            try {
                const response = await fetch(`https://viacep.com.br/ws/${cep}/json/`);
                const dados = await response.json();
                
                if (dados.erro) {
                    alert('CEP não encontrado!');
                } else {
                    document.getElementById('endereco').value = dados.logradouro || '';
                    document.getElementById('bairro_cidade').value = `${dados.bairro} - ${dados.localidade}/${dados.uf}`;
                    
                    // Salvar endereço localmente para não precisar digitar de novo (LocalStorage)
                    let enderecoSalvo = {
                        cep: cep,
                        rua: dados.logradouro,
                        bairro_cidade: `${dados.bairro} - ${dados.localidade}/${dados.uf}`
                    };
                    localStorage.setItem('endereco_parada', JSON.stringify(enderecoSalvo));
                }
            } catch (error) {
                alert('Erro ao buscar CEP. Tente novamente.');
            }
        }

        // Carregar endereço salvo ao abrir a página
        window.onload = function() {
            let enderecoSalvo = localStorage.getItem('endereco_parada');
            if (enderecoSalvo) {
                let dados = JSON.parse(enderecoSalvo);
                document.getElementById('cep').value = dados.cep;
                document.getElementById('endereco').value = dados.rua;
                document.getElementById('bairro_cidade').value = dados.bairro_cidade;
            }
        };

        // Enviar Pedido para o WhatsApp
        function finalizarPedido() {
            if (carrinho.length === 0) return alert('Seu carrinho está vazio!');

            let endereco = document.getElementById('endereco').value;
            let bairroCidade = document.getElementById('bairro_cidade').value;
            let taxa = parseFloat(document.getElementById('distancia').value);
            let pagamento = document.getElementById('pagamento').value;
            let total = parseFloat(document.getElementById('total').innerText.replace('R$', '').replace(',', '.'));
            
            // Montar mensagem
            let mensagem = "Olá! Gostaria de fazer um pedido:\n\n";
            carrinho.forEach(item => {
                mensagem += `- ${item.nome} (R$ ${item.preco.toFixed(2)})\n`;
            });
            mensagem += `\nSubtotal: R$ ${document.getElementById('subtotal').innerText}`;
            mensagem += `\nTaxa de Entrega: R$ ${taxa.toFixed(2)}`;
            mensagem += `\n*Total: R$ ${total.toFixed(2)}*`;
            mensagem += `\n\nForma de Pagamento: ${pagamento}`;
            mensagem += `\n\nEndereço de Entrega:`;
            mensagem += `\n${endereco}`;
            mensagem += `\n${bairroCidade}`;
            
            // Link do WhatsApp
            let numero = "5521991166886";
            let url = `https://wa.me/${numero}?text=${encodeURIComponent(mensagem)}`;
            
            window.open(url, '_blank');
        }
    </script>
